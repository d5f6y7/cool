const API = "https://app.meilianji.cn/web2/kananji/get_web_ip.php";
const CODESET = new Set(CAMS.map(c => c[3]));

async function freshUrl(code) {
  const body = new URLSearchParams({
    cameraIndexCode: code,
    camera_type: "0",
    user_id: "1",
    platformName: "ios",
  });
  const r = await fetch(API, {
    method: "POST",
    headers: {
      "Content-Type": "application/x-www-form-urlencoded",
      "User-Agent": "Mozilla/5.0",
    },
    body,
  });
  const d = await r.json();
  return (Array.isArray(d) && d[0] && d[0].ip) || "";
}

function playlist(origin, filterUnit) {
  let out = "#EXTM3U\n\n";
  let i = 0;
  for (const [unit, region, name, code, online] of CAMS) {
    if (!online) continue;
    if (filterUnit && unit !== filterUnit) continue;
    i += 1;
    const title = region ? `${region} ${name}` : name;
    out += `#EXTINF:-1 group-title="${unit}",${i}. ${title}\n${origin}/cam/${code}.m3u8\n\n`;
  }
  return out;
}

function playlistResponse(origin, filterUnit) {
  return new Response(playlist(origin, filterUnit), {
    headers: { "Content-Type": "application/vnd.apple.mpegurl; charset=utf-8" },
  });
}

export default {
  async fetch(request) {
    const url = new URL(request.url);
    const origin = url.origin;
    const m = url.pathname.match(/^\/cam\/(\d+)\.m3u8$/);
    if (m) {
      if (!CODESET.has(m[1])) return new Response("unknown camera", { status: 404 });
      const target = await freshUrl(m[1]);
      if (!target) return new Response("camera offline", { status: 502 });
      return Response.redirect(target, 302);
    }
    const tm = url.pathname.match(/^\/town\/(\d+)\.m3u$/);
    if (tm) {
      const idx = parseInt(tm[1], 10) - 1;
      if (idx < 0 || idx >= UNITS.length) return new Response("unknown town", { status: 404 });
      return playlistResponse(origin, UNITS[idx]);
    }
    if (url.pathname === "/" || url.pathname === "/all.m3u" || url.pathname === "/fl.m3u") {
      const q = url.searchParams.get("unit");
      if (q && UNITS.includes(q)) return playlistResponse(origin, q);
      return playlistResponse(origin, null);
    }
    if (url.pathname === "/towns" || url.pathname === "/towns.txt") {
      let out = "";
      UNITS.forEach((u, i) => { out += `${i+1}. ${u}  ${origin}/town/${i+1}.m3u\n`; });
      return new Response(out, { headers: { "Content-Type": "text/plain; charset=utf-8" } });
    }
    return new Response("not found", { status: 404 });
  },
};
