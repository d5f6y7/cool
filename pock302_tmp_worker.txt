// pock302 — 302-type permanent links hub (Eric's 302.pock77.de5.net)
// Each open fetches a fresh signed URL from the source API and 302-redirects.
const API = 'https://gzbn.gztv.com:7443/media-cloud-manage-app/liveChannel/queryLiveChannelList';

const GZ_TV = [
  ['zonghes', '广州综合频道'],
  ['xinwen', '广州新闻频道'],
  ['nanguodushi', '广州4K南国都市频道'],
];
const NN_TV = [
  ['南宁新闻综合', 'https://hls.nntv.cn/nnlive/XWZH_24.m3u8'],
  ['南宁文旅生活', 'https://hls.nntv.cn/nnlive/WLSH_24.m3u8'],
  ['南宁影视娱乐', 'https://hls.nntv.cn/nnlive/YSYL_244.m3u8'],
  ['南宁公共', 'https://hls.nntv.cn/nnlive/GGPD_24.m3u8'],
];
const GZ_FM = [
  ['fm962', '广州电台FM96.2'],
  ['fm1061', '广州电台FM106.1'],
  ['fm1027', '广州电台FM102.7'],
  ['fm880', '广州电台FM88.0'],
];

async function freshUrl(type, stream) {
  const r = await fetch(API + '?type=' + type, { headers: { 'User-Agent': 'Mozilla/5.0' } });
  if (!r.ok) return null;
  const d = await r.json();
  for (const c of (d.data || [])) {
    if (c.httpUrl && c.httpUrl.includes('/live/' + stream + '.m3u8')) return c.httpUrl;
  }
  return null;
}

function playlist(host, title, items, prefix) {
  let out = '#EXTM3U\n';
  for (const [stream, name] of items) out += '#EXTINF:-1,' + name + '\nhttps://' + host + '/' + prefix + '/' + stream + '.m3u8\n';
  return new Response(out, { headers: { 'Content-Type': 'application/vnd.apple.mpegurl; charset=utf-8' } });
}

export default {
  async fetch(req) {
    const u = new URL(req.url);
    const p = u.pathname;
    const host = u.host;
    if (p === '/' ) {
      return new Response('302 links hub\n/gztv.m3u - 广州电视\n/gzfm.m3u - 广州电台\n', { headers: { 'Content-Type': 'text/plain; charset=utf-8' } });
    }
    if (p === '/gztv.m3u') return playlist(host, 'gztv', GZ_TV, 'gz');
    if (p === '/gzfm.m3u') return playlist(host, 'gzfm', GZ_FM, 'gzfm');
    if (p === '/gz.m3u') {
      let out = '#EXTM3U\n';
      for (const [stream, name] of GZ_TV) out += '#EXTINF:-1 group-title="广州电视",' + name + '\nhttps://' + host + '/gz/' + stream + '.m3u8\n';
      for (const [stream, name] of GZ_FM) out += '#EXTINF:-1 group-title="广州电台",' + name + '\nhttps://' + host + '/gzfm/' + stream + '.m3u8\n';
      for (const [name, url] of NN_TV) out += '#EXTINF:-1 group-title="南宁电视",' + name + '\n' + url + '\n';
      return new Response(out, { headers: { 'Content-Type': 'application/vnd.apple.mpegurl; charset=utf-8' } });
    }
    let m = p.match(/^\/(gz|gzfm)\/([a-z0-9]+)\.m3u8$/);
    if (m) {
      const url = await freshUrl(m[1] === 'gz' ? 1 : 2, m[2]);
      if (!url) return new Response('upstream unavailable', { status: 502 });
      return Response.redirect(url, 302);
    }
    return new Response('not found', { status: 404 });
  }
};
