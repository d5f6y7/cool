// ---- MD5 (pure JS; Workers WebCrypto 不支持 MD5) ----
function md5(input) {
  const str = unescape(encodeURIComponent(input));
  const S = [
    7, 12, 17, 22, 7, 12, 17, 22, 7, 12, 17, 22, 7, 12, 17, 22,
    5, 9, 14, 20, 5, 9, 14, 20, 5, 9, 14, 20, 5, 9, 14, 20,
    4, 11, 16, 23, 4, 11, 16, 23, 4, 11, 16, 23, 4, 11, 16, 23,
    6, 10, 15, 21, 6, 10, 15, 21, 6, 10, 15, 21, 6, 10, 15, 21,
  ];
  const K = [];
  for (let i = 0; i < 64; i++) K[i] = Math.floor(Math.abs(Math.sin(i + 1)) * 4294967296) >>> 0;
  const bytes = [];
  for (let i = 0; i < str.length; i++) bytes.push(str.charCodeAt(i) & 0xff);
  const bitLen = bytes.length * 8;
  bytes.push(0x80);
  while (bytes.length % 64 !== 56) bytes.push(0);
  // 64-bit little-endian length (high part assumed 0 for our short inputs)
  let low = bitLen >>> 0, high = Math.floor(bitLen / 4294967296) >>> 0;
  for (let i = 0; i < 4; i++) { bytes.push(low & 0xff); low >>>= 8; }
  for (let i = 0; i < 4; i++) { bytes.push(high & 0xff); high >>>= 8; }
  let a0 = 0x67452301, b0 = 0xefcdab89, c0 = 0x98badcfe, d0 = 0x10325476;
  const F = (x, y, z) => (x & y) | (~x & z);
  const G = (x, y, z) => (x & z) | (y & ~z);
  const H = (x, y, z) => x ^ y ^ z;
  const I = (x, y, z) => y ^ (x | ~z);
  const rotl = (x, n) => ((x << n) | (x >>> (32 - n))) >>> 0;
  for (let off = 0; off < bytes.length; off += 64) {
    const M = [];
    for (let j = 0; j < 16; j++) {
      M[j] = (bytes[off + j * 4] | (bytes[off + j * 4 + 1] << 8) | (bytes[off + j * 4 + 2] << 16) | (bytes[off + j * 4 + 3] << 24)) >>> 0;
    }
    let A = a0, B = b0, C = c0, D = d0;
    for (let i = 0; i < 64; i++) {
      let f, g;
      if (i < 16) { f = F(B, C, D); g = i; }
      else if (i < 32) { f = G(B, C, D); g = (5 * i + 1) % 16; }
      else if (i < 48) { f = H(B, C, D); g = (3 * i + 5) % 16; }
      else { f = I(B, C, D); g = (7 * i) % 16; }
      const tmp = D;
      D = C;
      C = B;
      B = (B + rotl((A + f + K[i] + M[g]) >>> 0, S[i])) >>> 0;
      A = tmp;
    }
    a0 = (a0 + A) >>> 0; b0 = (b0 + B) >>> 0; c0 = (c0 + C) >>> 0; d0 = (d0 + D) >>> 0;
  }
  const wordHex = (n) => {
    let s = '';
    for (let i = 0; i < 4; i++) s += ((n >>> (i * 8)) & 0xff).toString(16).padStart(2, '0');
    return s;
  };
  return wordHex(a0) + wordHex(b0) + wordHex(c0) + wordHex(d0);
}

// ---- 上游接口 ----
function upstreamHeaders(liveId) {
  return {
    Origin: 'https://www.sztv.com.cn',
    Referer: liveId
      ? `https://www.sztv.com.cn/pindao/index.html?liveId=${encodeURIComponent(liveId)}`
      : 'https://www.sztv.com.cn/pindao/',
    'User-Agent': UA,
  };
}

async function hmacSha512Base64(secret, text) {
  const key = await crypto.subtle.importKey('raw', new TextEncoder().encode(secret), { name: 'HMAC', hash: 'SHA-512' }, false, ['sign']);
  const sig = await crypto.subtle.sign('HMAC', key, new TextEncoder().encode(text));
  let bin = '';
  for (const b of new Uint8Array(sig)) bin += String.fromCharCode(b);
  return btoa(bin);
}

let catalogCache = null; // { rows, expiresAt }
async function fetchChannelList() {
  const now = Date.now();
  if (catalogCache && catalogCache.expiresAt > now) return catalogCache.rows;
  const url = new URL(CATALOG_URL);
  const date = new Date(now).toUTCString();
  const nonce = crypto.randomUUID();
  const canonical = `x-date: ${date}\n@request-target: get ${url.pathname}\nhost: ${url.host}\nnonce: ${nonce}`;
  const signature = await hmacSha512Base64(WEB_HMAC_SECRET, canonical);
  const resp = await fetch(CATALOG_URL, {
    headers: {
      ...upstreamHeaders(),
      'X-Date': date,
      Nonce: nonce,
      Authorization: `hmac username="${WEB_HMAC_USER}", algorithm="hmac-sha512", headers="x-date @request-target host nonce", signature="${signature}"`,
      Accept: 'application/json',
    },
  });
  if (!resp.ok) throw new Error(`频道接口 HTTP ${resp.status}`);
  const payload = await resp.json();
  const rows = [];
  const seen = new Set();
  for (const item of Array.isArray(payload?.returnData) ? payload.returnData : []) {
    const sourceName = String(item?.extend?.refername || item?.name || '').trim();
    const name = NAME_OVERRIDES[sourceName];
    const liveId = String(item?.extend?.liveId || '').trim();
    const rate = Number(item?.extend?.liveRate?.[0] || 500);
    const id = String(item?.id || '');
    if (!name || !/^[A-Za-z0-9]{3,32}$/.test(liveId) || !/^\d{1,10}$/.test(id) || seen.has(id)) continue;
    seen.add(id);
    rows.push({ id, name, liveId, rate });
  }
  if (payload?.returnCode !== '0000' || !rows.length) throw new Error('频道接口返回异常');
  catalogCache = { rows, expiresAt: now + 4 * 60 * 60 * 1000 };
  return rows;
}

const liveKeyCache = new Map(); // liveId -> { value, expiresAt }
function decodeLiveKey(raw) {
  let value = String(raw ?? '').trim();
  if (value.startsWith('"')) { try { value = JSON.parse(value); } catch { /* fallthrough */ } }
  if (!value || value === '0') throw new Error('直播 Key 接口拒绝请求');
  const split = value.length - Math.floor(value.length / 2);
  const rotated = value.slice(split) + value.slice(0, split);
  const decoded = atob([...rotated].reverse().join(''));
  if (!/^[A-Za-z0-9]{3,32}$/.test(decoded)) throw new Error('直播 Key 返回格式异常');
  return decoded;
}
async function fetchLiveKey(row) {
  const cached = liveKeyCache.get(row.liveId);
  if (cached && cached.expiresAt > Date.now()) return cached.value;
  const seconds = Math.floor(Date.now() / 1000);
  const url = new URL(LIVE_KEY_URL);
  url.searchParams.set('t', String(seconds));
  url.searchParams.set('id', row.liveId);
  url.searchParams.set('token', md5(`${seconds}${row.liveId}${LIVE_TOKEN_SALT}`));
  url.searchParams.set('at', '1');
  const resp = await fetch(url.href, { headers: upstreamHeaders(row.liveId) });
  if (!resp.ok) throw new Error(`直播 Key 接口 HTTP ${resp.status}`);
  const value = decodeLiveKey(await resp.text());
  liveKeyCache.set(row.liveId, { value, expiresAt: Date.now() + 30 * 60 * 1000 });
  return value;
}

function signStreamUrl(rawUrl, nowMs) {
  const url = new URL(rawUrl);
  const expires = (Math.floor((nowMs ?? Date.now()) / 1000) + STREAM_TTL_SECONDS).toString(16);
  url.searchParams.set('sign', md5(`${CDN_SIGN_SECRET}${url.pathname}${expires}`));
  url.searchParams.set('t', expires);
  return url.href;
}

// ---- HLS 中转 ----
function rewritePlaylist(text, baseUrl, origin) {
  const out = [];
  for (const line of text.split(/\r?\n/)) {
    if (line.startsWith('#')) {
      out.push(line.replace(/URI="([^"]+)"/g, (m, uri) => {
        const abs = new URL(uri, baseUrl).href;
        return `URI="${origin}/p?u=${encodeURIComponent(abs)}"`;
      }));
    } else if (line.trim()) {
      const abs = new URL(line.trim(), baseUrl).href;
      out.push(`${origin}/p?u=${encodeURIComponent(abs)}`);
    } else {
      out.push(line);
    }
  }
  return out.join('\n');
}

async function fetchPlaylistAndRewrite(upstreamUrl, origin, liveId) {
  const signed = signStreamUrl(upstreamUrl);
  const resp = await fetch(signed, { headers: upstreamHeaders(liveId) });
  const text = await resp.text();
  if (!resp.ok) return new Response(text, { status: resp.status, headers: { 'content-type': resp.headers.get('content-type') || 'text/plain' } });
  return new Response(rewritePlaylist(text, signed, origin), {
    headers: { 'content-type': 'application/vnd.apple.mpegurl', 'cache-control': 'no-store' },
  });
}

export default {
  async fetch(request) {
    const url = new URL(request.url);
    const origin = url.origin;
    try {
      if (url.pathname === '/' || url.pathname === '/sztv.m3u') {
        const rows = await fetchChannelList();
        const lines = ['#EXTM3U'];
        for (const r of rows) {
          lines.push(`#EXTINF:-1,${r.name}`);
          lines.push(`${origin}/ch/${r.id}.m3u8`);
        }
        return new Response(lines.join('\n') + '\n', { headers: { 'content-type': 'application/x-mpegurl; charset=utf-8' } });
      }
      const chMatch = url.pathname.match(/^\/ch\/(\d{1,10})\.m3u8$/);
      if (chMatch) {
        const rows = await fetchChannelList();
        const row = rows.find((r) => r.id === chMatch[1]);
        if (!row) return new Response('channel not found', { status: 404 });
        const key = await fetchLiveKey(row);
        const unsigned = `https://${STREAM_HOSTS[0]}/${row.liveId}/${row.rate}/${key}.m3u8`;
        return fetchPlaylistAndRewrite(unsigned, origin, row.liveId);
      }
      if (url.pathname === '/p') {
        const target = url.searchParams.get('u');
        if (!target) return new Response('missing u', { status: 400 });
        const u = new URL(target);
        if (u.protocol !== 'https:' || !STREAM_HOSTS.includes(u.hostname)) return new Response('bad upstream', { status: 400 });
        u.searchParams.delete('sign');
        u.searchParams.delete('t');
        const signed = signStreamUrl(u.href);
        const resp = await fetch(signed, { headers: upstreamHeaders() });
        const type = resp.headers.get('content-type') || '';
        if (u.pathname.endsWith('.m3u8') || type.includes('mpegurl')) {
          const text = await resp.text();
          if (!resp.ok) return new Response(text, { status: resp.status, headers: { 'content-type': type || 'text/plain' } });
          return new Response(rewritePlaylist(text, signed, origin), {
            headers: { 'content-type': 'application/vnd.apple.mpegurl', 'cache-control': 'no-store' },
          });
        }
        return new Response(resp.body, {
          status: resp.status,
          headers: { 'content-type': type || 'application/octet-stream', 'cache-control': 'no-store' },
        });
      }
      if (url.pathname === '/debug') {
        const rows = await fetchChannelList();
        const first = rows[0];
        let keyOk = false, playlistStatus = 0;
        try {
          const key = await fetchLiveKey(first);
          keyOk = /^[A-Za-z0-9]{3,32}$/.test(key);
          const signed = signStreamUrl(`https://${STREAM_HOSTS[0]}/${first.liveId}/${first.rate}/${key}.m3u8`);
          const r = await fetch(signed, { headers: upstreamHeaders(first.liveId) });
          playlistStatus = r.status;
        } catch { /* reported below */ }
        return Response.json({ channels: rows.map((r) => ({ id: r.id, name: r.name })), keyOk, playlistStatus });
      }
      return new Response('not found', { status: 404 });
    } catch (error) {
      return new Response(`error: ${error?.message || String(error)}`, { status: 502 });
    }
  },
};
